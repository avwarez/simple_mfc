# Portability suite: does this branch build, link and run everywhere it
# claims to -- Windows and Linux, on x86, x64 and arm64?
#
# Two targets, both cheap, both compiler- and architecture-agnostic:
#
#   smfc_selfcontained  one translation unit per public header, each
#                       including only that header. A header that quietly
#                       depends on another one having been included first
#                       works in every build here and breaks in the first
#                       consumer that includes it alone -- which is how
#                       eMule includes most of them.
#
#   smfc_instantiate    every class template instantiated in full, plus the
#                       members no conformance case reaches, plus the
#                       static_asserts that pin the language standard and
#                       the Win32 integer widths on this target. See the
#                       banner in instantiate.cpp.
#
# See ../../.github/workflows/portability.yml, which runs both across the
# six platform/architecture pairs.

# One TU per header. Generated at configure time from what include/ holds:
# a header added later needs a re-run of cmake to be picked up, which is
# what the configure step of every CI job does anyway.
file(GLOB SIMPLE_MFC_PUBLIC_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/../../include/*.h")
list(SORT SIMPLE_MFC_PUBLIC_HEADERS)

set(SIMPLE_MFC_SELF_TUS)
foreach(header IN LISTS SIMPLE_MFC_PUBLIC_HEADERS)
    get_filename_component(header_stem "${header}" NAME_WE)
    get_filename_component(header_name "${header}" NAME)
    set(generated "${CMAKE_CURRENT_BINARY_DIR}/self_${header_stem}.cpp")
    file(WRITE "${generated}"
        "// generated by tests/portability/CMakeLists.txt -- do not edit\n"
        "// Is ${header_name} self-contained? Nothing else is included here.\n"
        "#include \"${header_name}\"\n"
        "int smfc_self_contained_${header_stem}() { return 0; }\n")
    list(APPEND SIMPLE_MFC_SELF_TUS "${generated}")
endforeach()

add_library(smfc_selfcontained STATIC ${SIMPLE_MFC_SELF_TUS})
target_link_libraries(smfc_selfcontained PRIVATE simple_mfc::simple_mfc)
target_compile_definitions(smfc_selfcontained PRIVATE UNICODE _UNICODE)

add_executable(smfc_instantiate instantiate.cpp)
target_link_libraries(smfc_instantiate PRIVATE simple_mfc::simple_mfc)
target_compile_definitions(smfc_instantiate PRIVATE UNICODE _UNICODE)

# Running it is part of the check: the static_asserts and the missing
# definitions are caught by the compiler and the linker, but "the binary
# this architecture produced actually starts and returns 0" is a fact only
# an execution establishes -- and on the two Windows-on-ARM configurations
# the runner is the target, so it can.
add_test(NAME portability_instantiate COMMAND smfc_instantiate)
set_tests_properties(portability_instantiate PROPERTIES TIMEOUT 120)
